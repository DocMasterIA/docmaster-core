# ============================================
# COMANDOS DO RAG SERVICE - DocMaster IA
# ============================================

# IMPORTANTE: Sempre executar docker compose de onde estiver
# o docker-compose.yml principal (ex: ~/ ou ~/docmaster-core)

# --------------------------------------------
# CLASSIFICAÇÃO POR CAMINHO (não por conteúdo)
# --------------------------------------------
# O tipo do documento é INFERIDO pela pasta:
#   /data/treinamento/TERMO DE REFERENCIA/... → termo_referencia
#   /data/treinamento/ETP/... → etp
#   /data/treinamento/DFD/... → dfd
#
# Isso é mais confiável que tentar detectar pelo texto.

# --------------------------------------------
# SUBIR SERVIÇOS
# --------------------------------------------

# Subir todos os serviços
docker compose up -d

# Subir apenas o rag_service
docker compose up -d rag_service

# Reiniciar rag_service
docker compose up -d rag_service --force-recreate

# Ver logs
docker logs rag_service -f

# --------------------------------------------
# TESTAR CLASSIFICAÇÃO (sem indexar)
# --------------------------------------------

# Mostra tipo inferido pelo caminho + seções extraídas
curl -X POST http://localhost:8004/api/rag/classify \
  -H "Content-Type: application/json" \
  -d '{"path": "/data/treinamento/TERMO DE REFERENCIA/TERMO DE REFERENCIA1.pdf"}'

# Forçar tipo específico no teste
curl -X POST http://localhost:8004/api/rag/classify \
  -H "Content-Type: application/json" \
  -d '{"path": "/data/treinamento/DFD/arquivo.pdf", "document_type": "dfd"}'

# --------------------------------------------
# INGERIR UM ARQUIVO
# --------------------------------------------

# Tipo inferido pelo caminho da pasta
curl -X POST http://localhost:8004/api/rag/smart-ingest/file \
  -H "Content-Type: application/json" \
  -d '{"path": "/data/treinamento/TERMO DE REFERENCIA/TERMO DE REFERENCIA1.pdf"}'

# Tipo informado explicitamente (se a pasta não indicar)
curl -X POST http://localhost:8004/api/rag/smart-ingest/file \
  -H "Content-Type: application/json" \
  -d '{"path": "/data/outros/documento.pdf", "document_type": "dfd"}'

# --------------------------------------------
# INGERIR PASTA INTEIRA
# --------------------------------------------

# Termos de Referência
curl -X POST http://localhost:8004/api/rag/smart-ingest/folder \
  -H "Content-Type: application/json" \
  -d '{"folder_path": "/data/treinamento/TERMO DE REFERENCIA"}'

# ETPs
curl -X POST http://localhost:8004/api/rag/smart-ingest/folder \
  -H "Content-Type: application/json" \
  -d '{"folder_path": "/data/treinamento/ETP"}'

# DFDs
curl -X POST http://localhost:8004/api/rag/smart-ingest/folder \
  -H "Content-Type: application/json" \
  -d '{"folder_path": "/data/treinamento/DFD"}'

# Lei 14.133 (para verificação de conformidade)
# 1. Coloque o PDF da lei em: treinamento/LEI_14_133/
# 2. Ingerir:
curl -X POST http://localhost:8004/api/rag/smart-ingest/folder \
  -H "Content-Type: application/json" \
  -d '{"folder_path": "/data/treinamento/LEI_14_133"}'

# --------------------------------------------
# VERIFICAR CONFORMIDADE (Lei 14.133)
# --------------------------------------------

# Verifica se texto gerado está em conformidade com a lei
curl -X POST http://localhost:8004/api/rag/check-compliance \
  -H "Content-Type: application/json" \
  -d '{"text": "O prazo de vigência será de 12 meses, podendo ser prorrogado por igual período."}'

# Com mais trechos da lei para comparação
curl -X POST http://localhost:8004/api/rag/check-compliance \
  -H "Content-Type: application/json" \
  -d '{"text": "O contrato terá vigência de 5 anos.", "top_k": 8}'

# --------------------------------------------
# CONSULTAR (QUERY)
# --------------------------------------------

# Pergunta simples
curl -X POST http://localhost:8004/api/rag/query \
  -H "Content-Type: application/json" \
  -d '{"query": "qual o objeto do contrato?"}'

# Só recuperar documentos (sem gerar resposta)
curl -X POST http://localhost:8004/api/rag/query \
  -H "Content-Type: application/json" \
  -d '{"query": "cláusula de justificativa", "generate_answer": false}'

# Buscar em coleção específica
curl -X POST http://localhost:8004/api/rag/query \
  -H "Content-Type: application/json" \
  -d '{"query": "objeto", "collection": "minha_colecao"}'

# --------------------------------------------
# VER RESUMO DA BASE
# --------------------------------------------

curl http://localhost:8004/api/rag/summary

# --------------------------------------------
# LISTAR COLEÇÕES
# --------------------------------------------

curl http://localhost:8004/api/rag/collections

# --------------------------------------------
# DELETAR COLEÇÃO
# --------------------------------------------

curl -X DELETE http://localhost:8004/api/rag/collections/docmaster_docs

# --------------------------------------------
# HEALTH CHECK
# --------------------------------------------

curl http://localhost:8004/api/rag/health
