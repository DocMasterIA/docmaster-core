# --------------------------------------------
# CLASSIFICAÇÃO POR CAMINHO (não por conteúdo)
# --------------------------------------------
# O tipo do documento é INFERIDO pela pasta:
#   /data/treinamento/TERMO DE REFERENCIA/... → termo_referencia
#   /data/treinamento/ETP/... → etp
#   /data/treinamento/DFD/... → dfd
#
# Isso é mais confiável que tentar detectar pelo texto.

# --------------------------------------------
# SUBIR SERVIÇOS
# --------------------------------------------

# Subir todos os serviços
docker compose up -d

# Subir apenas o rag_service
docker compose up -d rag_service

# Reiniciar rag_service
docker compose up -d rag_service --force-recreate

# Ver logs
docker logs rag_service -f

# --------------------------------------------
# TESTAR CLASSIFICAÇÃO (sem indexar)
# --------------------------------------------

# Mostra tipo inferido pelo caminho + seções extraídas
curl -X POST http://localhost:8004/api/rag/classify \
  -H "Content-Type: application/json" \
  -d '{"path": "/data/treinamento/TERMO DE REFERENCIA/TERMO DE REFERENCIA1.pdf"}'

# Forçar tipo específico no teste
curl -X POST http://localhost:8004/api/rag/classify \
  -H "Content-Type: application/json" \
  -d '{"path": "/data/treinamento/DFD/arquivo.pdf", "document_type": "dfd"}'

# ---- INSPECIONAR: diagnosticar erros (pt, inglês, palavras cortadas) ----

  -d '{"user_prompt": "gerar ETP para coleta seletiva"}' | jq . > inspecao.json
# Decodificar DOCX: jq -r '.docx_base64' inspecao.json | base64 -d > documento_inspecao.docx
# Ver textos brutos: jq '.section_texts' inspecao.json

# --------------------------------------------
# RENDER-DOCUMENT (JSON → DOCX/PDF)
# --------------------------------------------
# Gera DOCX e PDF a partir de JSON (structure + section_texts).
# Use o JSON retornado por generate-document ou monte manualmente.

# Baixar DOCX (padrão)
curl -X POST http://localhost:8004/api/rag/render-document \
  -H "Content-Type: application/json" \
  -d '{
    "structure": {
      "document_type": "estudo_tecnico_preliminar",
      "root_sections": [
        {"key": "objeto", "title": "DO OBJETO", "content_type": "text", "required": true, "children": []},
        {"key": "necessidade", "title": "DA NECESSIDADE", "content_type": "text", "required": true, "children": []},
        {"key": "estimativa_custos", "title": "DA ESTIMATIVA DE CUSTOS", "content_type": "table", "required": true,
         "table_schema": {"columns": [{"key": "item", "label": "Item"}, {"key": "quantidade", "label": "Quantidade"}, {"key": "valor_unitario", "label": "Valor Unitário"}]},
         "children": []}
      ]
    },
    "section_texts": {
      "objeto": "Contratação de serviços de coleta seletiva de resíduos sólidos.",
      "necessidade": "Necessidade de realizar a coleta seletiva conforme legislação ambiental."
    },
    "table_data": {
      "estimativa_custos": [{"item": "Coleta mensal", "quantidade": "12", "valor_unitario": "R$ 5.000,00"}]
    }
  }' -o documento.docx

# Retornar JSON com caminhos (docx_path, pdf_path se LibreOffice instalado)
curl -X POST "http://localhost:8004/api/rag/render-document?format=json" \
  -H "Content-Type: application/json" \
  -d '{"structure": {"document_type": "termo_referencia", "root_sections": [{"key": "objeto", "title": "DO OBJETO", "content_type": "text", "required": true, "children": []}]}, "section_texts": {"objeto": "Objeto do contrato."}}'
